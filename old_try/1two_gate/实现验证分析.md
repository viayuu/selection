#### 路径统计特征计算 (1two_gate/features.py:129-151)
```python
def tsp_tour_features(coords, tour):
    """
    coords: Tensor[batch, n, 2]
    tour:   Tensor[batch, n]  (permutation of 0..n-1)
    returns: Tensor[batch, d]
    """
    # 计算路径段长度
    ordered = coords.gather(1, tour.unsqueeze(-1).expand(-1, -1, 2))
    rolled = ordered.roll(shifts=-1, dims=1)
    seg = ((ordered - rolled) ** 2).sum(dim=-1).sqrt()

    # 路径统计特征
    mean = seg.mean(dim=1, keepdim=True)
    std = seg.std(dim=1, keepdim=True)
    minv = seg.min(dim=1, keepdim=True)
    maxv = seg.max(dim=1, keepdim=True)

    return torch.cat([mean, std, minv, maxv], dim=1)
```

```python
def run_initializers_by_action(coords, actions, zoo: list[Initializer]) -> SolutionBatch:
    """根据动作执行对应的初始化器"""
    groups = _gather_by_action(actions, len(zoo))
    for action_id, idx in enumerate(groups):
        if idx.numel() == 0:
            continue
        sub = zoo[action_id].solve(coords[idx])  # 执行对应求解器
        out_tour[idx] = sub.tour
        out_len[idx] = sub.length
    return SolutionBatch(tour=out_tour, length=out_len)

def run_iterators_by_action(coords, actions, sol0: SolutionBatch, zoo: list[Iterator]) -> SolutionBatch:
    """根据动作执行对应的迭代器"""
    groups = _gather_by_action(actions, len(zoo))
    for action_id, idx in enumerate(groups):
        if idx.numel() == 0:
            continue
        sub0 = SolutionBatch(tour=sol0.tour[idx], length=sol0.length[idx])
        sub1 = zoo[action_id].improve(coords[idx], sub0)  # 执行对应迭代器
        out_tour[idx] = sub1.tour
        out_len[idx] = sub1.length
    return SolutionBatch(tour=out_tour, length=out_len)
```

### 14.2 训练循环（每个step）
```python
for step in range(1, cfg.train_steps + 1):
    # Step 1: 生成训练数据
    coords = _make_tsp_batch(cfg.batch_size, cfg.problem_size, cfg.device)

    # Step 2: Gate 1 - 特征提取和动作选择
    feat1 = tsp_instance_features(coords, cfg=feat_cfg, encoder=policy.encoder)
    dist1 = Categorical(logits=policy.gate1(feat1))
    a1 = dist1.sample()  # 采样动作
    logp1 = dist1.log_prob(a1)  # 记录对数概率

    # Step 3: Gate 1 - 执行初始化器
    sol0 = run_initializers_by_action(coords, a1, init_zoo)

    # Step 4: Gate 2 - 特征提取（包含初始解信息）
    feat2 = tsp_gate2_features(coords, sol0.tour, sol0.length, cfg=feat_cfg, encoder=policy.encoder)
    dist2 = Categorical(logits=policy.gate2(feat2))
    a2 = dist2.sample()  # 采样动作
    logp2 = dist2.log_prob(a2)  # 记录对数概率

    # Step 5: Gate 2 - 执行迭代器
    sol1 = run_iterators_by_action(coords, a2, sol0, iter_zoo)

    # Step 6: 奖励计算
    length = sol1.length
    reward = -length  # reward = -length

    # Step 7: Baseline计算（根据配置选择模式）
    if cfg.baseline == "batch_mean":
        baseline_length = length.mean()
        baseline = -baseline_length
    elif cfg.baseline == "critic_batch_mean":
        baseline_length = length.mean()
        baseline_reward = -baseline_length
        value_pred = policy.value(feat1).squeeze(-1)
        baseline = baseline_reward + value_pred
        critic_loss = ((reward - baseline_reward - value_pred) ** 2).mean()

    # Step 8: 优势函数计算
    adv = reward - baseline

    # Step 9: 损失函数计算
    actor_loss = -((logp1 + logp2) * adv.detach()).mean()
    loss = actor_loss + cfg.critic_coef * critic_loss

    # Step 10: 网络更新
    optimizer.zero_grad(set_to_none=True)
    loss.backward()
    optimizer.step()

    # Step 11: 日志记录
    if step % cfg.log_every == 0:
        # 记录训练指标
        print(f"[train step {step}] loss={loss.item():.4f} ...")
```
