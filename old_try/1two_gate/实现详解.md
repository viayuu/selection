# 双层强化学习TSP求解器选择框架 - 完整实现分析

> **项目概述**：这是一个基于Actor-Critic强化学习的双层求解器选择框架，通过学习为每个TSP实例动态选择最优的初始化器和迭代器组合，以实现超越单个求解器的性能。
>
> **核心创新**：将求解器选择分解为两个独立阶段——初始化选择（Gate1）和迭代优化选择（Gate2），每个阶段都有独立的策略网络和Critic网络。

### 1.3 双层设计的必要性

**为什么需要双层架构？**

1. **不同阶段的需求不同**
   - 初始化阶段：需要生成一条完整的Hamiltonian回路
   - 迭代阶段：需要在给定初始解的基础上进行局部改进

2. **求解器能力互补**
   - POMO擅长并行初始化，但无法直接改进现有解
   - RRC需要初始解，但能显著改进解的质量
   - 2-opt是确定性的，不依赖神经网络

3. **训练效率**
   - 分离两个决策可以减少动作空间
   - Gate1只需考虑初始化质量
   - Gate2可以根据初始解状态做更精细的选择

```python
def tsp_instance_features(coords, *, cfg: FeatureConfig, encoder=None, mask=None):
    """
    输入: coords [B, N, 2] - TSP实例坐标
    输出: features [B, d] - 实例特征表示

    流程:
    1. 图编码器提取节点嵌入
    2. 通过注意力机制聚合节点信息
    3. 池化得到图级表示
    4. 拼接问题规模N
    """
    if cfg.extractor == "paper":
        # 使用论文风格的层次化图编码器
        graph_emb = encoder(coords, mask)  # [B, 2*embedding_dim] 或 [B, embedding_dim]
        feats = [graph_emb]
        if cfg.paper_include_scale:
            scale = torch.full((coords.size(0), 1), float(coords.size(1)), ...)
            feats.append(scale)  # 添加规模特征
        return torch.cat(feats, dim=1)
```

**输出维度计算**：
- 层次化编码器 (pooling=True): `2 * 128 + 1 = 257` 维
- 简单编码器 (pooling=False): `128 + 1 = 129` 维

#### 2.2.3 Gate2特征：实例+初始解特征

```python
def tsp_gate2_features(coords, tour0, tour0_length, *, cfg: FeatureConfig, encoder=None, mask=None):
    """
    输入:
        coords [B, N, 2] - TSP实例坐标
        tour0 [B, N] - 初始解（节点访问顺序）
        tour0_length [B] - 初始解长度
    输出: features [B, d] - 条件特征表示

    特征组成:
    1. 实例特征（与Gate1相同）
    2. 初始解长度 length0
    3. Tour统计特征（边长度统计）
    """
    base = tsp_instance_features(coords, cfg=cfg, encoder=encoder, mask=mask)

    extras = []
    if cfg.use_length0:
        extras.append(tour0_length[:, None])  # 添加长度特征

    if cfg.use_tour_stats:
        extras.append(tsp_tour_features(coords, tour0))  # 添加tour统计

    return torch.cat([base, *extras], dim=1)
```

**Tour统计特征**：
```python
def tsp_tour_features(coords, tour):
    """
    计算tour的边长度统计特征
    - 边长度的均值
    - 边长度的标准差
    - 最小边长度
    - 最大边长度
    """
    ordered = coords.gather(1, tour.unsqueeze(-1).expand(-1, -1, 2))
    rolled = ordered.roll(shifts=-1, dims=1)
    seg = ((ordered - rolled) ** 2).sum(dim=-1).sqrt()

    mean = seg.mean(dim=1, keepdim=True)
    std = seg.std(dim=1, keepdim=True)
    minv = seg.min(dim=1, keepdim=True).values
    maxv = seg.max(dim=1, keepdim=True).values

    return torch.cat([mean, std, minv, maxv], dim=1)
```

#### 2.3.4 神经网络初始化器：NeuralARInitializer

```python
class NeuralARInitializer(Initializer):
    """
    包装EasyNCO的自回归初始化方法（POMO/AM/LEHD/ELG）

    工作流程:
    1. 调用EasyNCO的initialization.run(...)
    2. 环境执行自回归解码，生成多条tour（POMO并行）
    3. 选择reward最高（长度最短）的tour作为输出
    """
    def solve(self, coords) -> SolutionBatch:
        with TorchNoGrad():
            state_td, _out = self.initialization.run(
                self.env,
                coords,
                self.decoder_strategy,  # "greedy" or "sampling"
                "eval",
            )

        # 提取结果
        reward = state_td.get("reward")      # [B, pomo]
        selected_node_list = state_td["selected_node_list"]  # [B, pomo, N]

        # 选择每个实例的最佳tour
        best_idx = reward.max(dim=1).indices  # [B]
        b = torch.arange(coords.size(0), device=coords.device)
        best_tour = selected_node_list[b, best_idx]  # [B,N]
        best_length = (-reward[b, best_idx]).to(coords.dtype)

        return SolutionBatch(tour=best_tour, length=best_length)
```

**POMO并行机制**：
- `pomo_size`：并行生成的tour数量（默认=problem_size）
- 每个实例生成POMO条tour，选择最好的作为输出
- 这是POMO论文的核心创新之一

```python
    传统2-opt局部搜索（确定性，无需神经网络）

    算法流程:
    1. 对于每条边对(i,j)
    2. 计算交换后的长度变化
    3. 如果改善，则接受交换
    4. 重复直到无法改进或达到max_iterations

    def improve(self, coords, sol0: SolutionBatch) -> SolutionBatch:
        from EasyNCO.neural_solvers.methods.difusco.util import two_opt_refine
```

```python
class RRCLIHStyleLEHDIterator(Iterator):
    """
    LEHD的RRC（Random Re-Construction）迭代优化

    核心思想: 破坏-修复
    1. 随机选择一条子路径并删除
    2. 使用LEHD策略重新构建该子路径
    3. 如果重建后的解更好，则接受；否则保持原解
    4. 重复max_steps次
    """
    def improve(self, coords, sol0: SolutionBatch) -> SolutionBatch:
        best_selected_node_list = sol0.tour[:, None, :]  # [B,1,N]

        for _ in range(self.max_steps):
            # 步骤1：破坏解（随机选择子路径删除）
            partial_len, first_node_index, subpath_length, solution_copy = \
                self._destroy_solution(problems, best_selected_node_list)

            prev_length = partial_len

            # 步骤2：修复解（使用LEHD策略重建子路径）
            reset_td = self.env.reset()
            self.policy.pre_forward(reset_td)
            state_td = self.env.pre_step()

            done = False
            current_step = 0
            while not done:
                if current_step == 0:
                    selected = self.env.solution[:, :, -1]
                elif current_step == 1:
                    selected = self.env.solution[:, :, 0]
                else:
                    selected = self.policy(state_td)  # LEHD策略选择下一个节点

                next_td["action"] = selected
                current_step += 1
                state_td = self.env.step(next_td)
                done = state_td["done"].all()

            repaired_sub_solution = torch.roll(self.env.selected_node_list, shifts=-1, dims=2)
            repaired_length = -state_td["reward"]  # [B,1]

            # 步骤3：接受或拒绝
            repaired_full_solution = self._accept_repaired_solution(
                repaired_sub_solution,
                prev_length,
                repaired_length,
                first_node_index,
                subpath_length,
                solution_copy,
            )

            best_selected_node_list = repaired_full_solution

        best_tour = best_selected_node_list.squeeze(1)
        best_length = self.env._get_travel_distance(...)

        return SolutionBatch(tour=best_tour, length=best_length)
```

```python
class DACTIterator(Iterator):
    """
    DACT (Deep ACT) 迭代优化

    核心思想: 学习性2-opt
    - 使用神经网络预测哪个2-opt交换最有可能改善解
```

**后继表示 vs 排列表示**：
```python
def _perm_to_succ(tour):
    """
    排列 → 后继
    tour: [0, 2, 1, 3]  表示访问顺序 0→2→1→3→0
    succ:  [2, 1, 3, 0]  表示succ[i]是i的下一个节点
    """
    nxt = tour.roll(shifts=-1, dims=1)
    succ = torch.empty_like(tour)
    succ.scatter_(1, tour, nxt)
    return succ

def _succ_to_perm(succ, start_node=0):
    """
    后继 → 排列
    从start_node开始，顺着succ链构建tour
    """
    tour = torch.empty((bsz, n), dtype=torch.long)
    cur = torch.full((bsz,), int(start_node))
    for i in range(n):
        tour[:, i] = cur
        cur = succ.gather(1, cur.view(-1, 1)).squeeze(1)
    return tour
```

#### 2.3.10 按动作运行求解器

```python
def run_initializers_by_action(coords, actions, zoo: list[Initializer]) -> SolutionBatch:
    """
    根据batch中每个实例的动作，选择对应的初始化器运行

    输入:
        coords [B,N,2] - TSP实例
        actions [B] - 动作（0~len(zoo)-1）
        zoo - 初始化器列表

    流程:
    1. 按动作将batch分组
    2. 对每组调用对应的初始化器
    3. 合并结果
    """
    batch_size = coords.size(0)
    out_tour = torch.zeros((batch_size, problem_size), ...)
    out_len = torch.zeros((batch_size,), ...)

    # 分组
    groups = _gather_by_action(actions, len(zoo))
    for action_id, idx in enumerate(groups):
        if idx.numel() == 0:
            continue
        # 运行对应的初始化器
        sub = zoo[action_id].solve(coords[idx])
        out_tour[idx] = sub.tour
        out_len[idx] = sub.length

    return SolutionBatch(tour=out_tour, length=out_len)
```

```python
class Naive_Encoder(nn.Module):
    """
    非层次化图编码器

    结构:
    1. 坐标嵌入 (2/3 → embedding_dim)
    2. 多层图注意力
    3. 平均池化得到图表示
    """
    def __init__(self, **model_params):
        super().__init__()
        self.model_params = model_params
        embedding_dim = model_params["embedding_dim"]

        # 节点嵌入层
        if model_params["problem_type"] == "TSP":
            self.embedding = nn.Linear(2, embedding_dim)
        elif model_params["problem_type"] == "CVRP":
            self.embedding = nn.Linear(3, embedding_dim)

        # 多层图注意力
        encoder_layer_num = model_params["encoder_layer_num"] * model_params["block_num"]
        self.layers = nn.ModuleList([EncoderLayer(**model_params) for _ in range(encoder_layer_num)])

    def forward(self, points, mask):
        embs = self.embedding(points)  # [B,N,2] → [B,N,d]

        # 多层注意力更新
        for layer in self.layers:
            embs = layer(embs, mask=mask)

        # 平均池化
        emb_mask = torch.where(mask == float("-inf"), 0, 1)
        embs = embs * emb_mask[:, :, None]
        graph_emb = embs.sum(dim=1) / emb_mask.sum(-1)[:, None]

        return graph_emb  # [B,d]
```

#### 2.5.2 Encoder_h：层次化图编码器

```python
class Encoder_h(nn.Module):
    """
    层次化图编码器（论文核心创新）

    关键创新: 可微分图池化
    - 通过学习选择代表性节点
    - 逐层减少节点数量
    - 聚合不同尺度的特征
    """
    def __init__(self, **model_params):
        super().__init__()
        self.model_params = model_params

        # 嵌入层
        self.embedding = nn.Linear(2, model_params["embedding_dim"])

        # 多个编码器块（每个块包含：注意力 → 池化）
        self.blocks = nn.ModuleList([Encoder_block_h(**model_params)
                                      for _ in range(model_params["block_num"])])
        self.nonlinear = nn.GELU()

    def forward(self, data, mask):
        out = self.embedding(data)  # [B,N,2] → [B,N,d]

        i = 0
        graph_emb_h = 0
        for block in self.blocks:
            # 每个块返回：当前层图表示、池化后节点嵌入、更新后掩码
            graph_emb, out, mask = block(out, mask, i)
            graph_emb_h += graph_emb  # 累积多尺度特征
            i += 1

        # 最终层Readout
        mean_emb = self.masked_mean(out, mask)
        max_emb = self.masked_max(out, mask)
        graph_emb = self.nonlinear(torch.cat((mean_emb, max_emb), dim=1))
        graph_emb_h += graph_emb

        return graph_emb_h  # [B, 2*d]（均值+最大值拼接后非线性变换）
```

**层次化池化详解**（Encoder_block_h）：

```python
class Encoder_block_h(nn.Module):
    def forward(self, embs, mask, _i):
        # 步骤1: 多层注意力更新
        for layer in self.layers:
            embs = layer(embs, mask)

        # 步骤2: Readout当前层特征
        mean_emb = self.masked_mean(embs, mask)
        max_emb = self.masked_max(embs, mask)
        graph_emb = self.nonlinear(torch.cat((mean_emb, max_emb), dim=1))

        # 步骤3: 可微分图池化
        score_embs = self.layer_score(embs, mask)  # 计算节点重要性分数
        scores = self.act(self.p(score_embs)).squeeze(-1)  # tanh激活
        scores = scores + mask

        # 为每个batch样本选择top-k节点
        selected_embs_list = []
        for b in range(embs.shape[0]):
            num_valid = int((mask[b] == 0).sum())
            k = max(1, int(num_valid * self.model_params["downsample_ratio"]))  # 池化比例
            score, ind = scores[b].topk(k, dim=-1, largest=True)

            # 关键：将分数融入节点嵌入，使池化可微分
            selected_emb = embs[b].take_along_dim(ind[:, None].expand(-1, embs.shape[-1]), dim=0)
            selected_emb = selected_emb + score[:, None]  # ← 可微性的关键
            selected_embs_list.append(selected_emb)

        # padding并返回
        selected_embs, selected_mask = self.padding_concate(selected_embs_list)

        return graph_emb, selected_embs, selected_mask
```

**可微分图池化的关键**：
1. **重要性评分**：通过注意力网络计算每个节点的重要性
2. **Top-K选择**：选择最重要的k个节点
3. **分数融入**：`selected_emb = selected_emb + score[:, None]`
4. **可微性**：选择的操作不可微，但分数融入使得梯度可以回传

#### 2.5.3 EncoderLayer：图注意力层

```python
class EncoderLayer(nn.Module):
    """
    Transformer风格的图注意力层

    结构:
    1. 多头自注意力
    2. 残差连接 + ReZero归一化
    3. 前馈网络
    4. 残差连接 + ReZero归一化
    """
    def forward(self, input1, mask=None, kv=None):
        # 多头注意力
        q = reshape_by_heads(self.Wq(input1), head_num=self.model_params["head_num"])
        k = reshape_by_heads(self.Wk(kv if kv else input1), head_num=self.model_params["head_num"])
        v = reshape_by_heads(self.Wv(kv if kv else input1), head_num=self.model_params["head_num"])

        out_concat = multi_head_attention(q, k, v, rank2_ninf_mask=mask)
        multi_head_out = self.multi_head_combine(out_concat)

        # 残差 + ReZero归一化
        out1 = self.addAndNormalization1(input1, multi_head_out)

        # 前馈网络
        out2 = self.feedForward(out1)

        # 残差 + ReZero归一化
        out3 = self.addAndNormalization2(input1, out2)

        return out3
```

**ReZero归一化**：
```python
class Add_And_Normalization_Module(nn.Module):
    def forward(self, input1, input2):
        if isinstance(self.norm, nn.Parameter):  # ReZero模式
            back_trans = input1 + self.norm * input2  # α * input2
        else:
            back_trans = input1 + input2  # 简单残差
        return back_trans
```

**优势**：
- 训练更稳定（相比LayerNorm）
- 参数更少（只有一个可学习标量α）
- 允许网络逐渐学习残差连接的强度

```python
    核心思想:
    1. 按照概率排序边 (adjacency / distance)
    2. 使用并查集逐步构建tour
    3. 确保最终形成单条Hamiltonian回路
    """
    def merge_cython(coords, adj_mat):
        import numpy as np

        n = int(points.shape[0])
        dist = np.linalg.norm(points[:, None, :] - points[None, :, :], axis=-1)
        np.fill_diagonal(dist, np.inf)

        # 按重要性排序边
        sorted_edges = np.argsort((-adj / dist).reshape(-1))

        # 并查集数据结构
        route_begin = np.arange(n, dtype=np.int64)
        route_end = np.arange(n, dtype=np.int64)

        A = np.zeros((n, n), dtype=np.float64)
        merge_count = 0

        for edge in sorted_edges:
            i = int(edge // n)
            j = int(edge % n)

            begin_i = find_begin(i)
            end_i = find_end(i)
            begin_j = find_begin(j)
            end_j = find_end(j)

            # 检查是否可以合并
            if begin_i == begin_j:
                continue
            if i != begin_i and i != end_i:
                continue
            if j != begin_j and j != end_j:
                continue

            # 合并路径
            A[j, i] = 1.0
            A[i, j] = 1.0
            merge_count += 1

            # 更新并查集
            if i == begin_i and j == end_j:
                route_begin[begin_i] = begin_j
                route_end[end_j] = end_i
            elif i == end_i and j == begin_j:
                route_begin[begin_j] = begin_i
                route_end[end_i] = end_j
            # ... 其他情况

            if merge_count == n - 1:
                break

        # 闭合回路
        final_begin = find_begin(0)
        final_end = find_end(0)
        A[final_end, final_begin] = 1.0
        A[final_begin, final_end] = 1.0
```

#### 2.7.2 单实例生成

```python
def _generate_tsp_instance_gaussian(problem_size, params, rs):
    """
    生成单个高斯混合TSP实例

    流程:
    1. 采样混合成分数 num_modes ~ U[0, 15]
    2. 如果 num_modes=0，退化为均匀分布
    3. 否则，为每个成分采样：
       - 混合比例
       - 节点数量（多项式分布）
       - 协方差矩阵
       - 中心点
    4. 合并所有成分的节点
    5. MinMax缩放到[0,1]×[0,1]
    """
    # 采样混合成分数
    if params.num_modes_lower == params.num_modes_upper:
        num_modes = int(params.num_modes_lower)
    else:
        num_modes = int(rs.randint(params.num_modes_lower, params.num_modes_upper))

    if num_modes == 0:
        return _generate_tsp_instance_uniform(problem_size, rs)

    # 采样混合比例和节点分配
    mix_proportion = rs.rand(num_modes)
    nums = rs.multinomial(int(problem_size), mix_proportion / mix_proportion.sum())

    xy_parts = []
    for num in nums:
        num = int(num)
        if num <= 0:
            continue

        # 采样协方差矩阵
        if params.no_cov:
            var = float(rs.uniform(params.var_lower, params.var_upper))
            cov = [[var, 0.0], [0.0, var]]
        else:
            var_x = float(rs.uniform(params.var_lower, params.var_upper))
            var_y = float(rs.uniform(params.var_lower, params.var_upper))
            cov_xy = float(rs.uniform(-np.sqrt(var_x * var_y), np.sqrt(var_x * var_y)))
            cov = [[var_x, cov_xy], [cov_xy, var_y]]

        # 采样中心
        center = rs.uniform(params.center_lower, params.center_upper, size=(1, 2))

        # 生成节点
        nxy = rs.multivariate_normal(mean=center.squeeze(), cov=cov, size=(num,))
        xy_parts.append(nxy)

    # 合并并缩放
    xy = np.concatenate(xy_parts, axis=0)
    xy = _minmax_scale(xy)
    return xy.astype("float32")
```

```python
    加载求解器策略的checkpoint

    支持的格式:
    - Lightning: {"state_dict": {...}}
    - Author格式: {"model_state_dict": {...}}

    - 不同保存方式可能添加不同的前缀（"policy.", "module.", "model."）
    - 尝试多种前缀组合，选择匹配最好的
    """
    # 获取目标模型的state_dict
    target = policy
    get_model = getattr(policy, "_get_model", None)
    if callable(get_model):
        target = get_model()

            state = ckpt["state_dict"]
        elif "model_state_dict" in ckpt:
            state = ckpt["model_state_dict"]
        else:
            state = ckpt

        (),
        ("policy.",),
        ("module.",),
        ("policy.", "module."),
        ("model.",),  # 注意：DIFUSCO的模型参数真的在"model.*"下
```

```python
    feat1 = tsp_instance_features(coords, cfg=feat_cfg, encoder=policy.encoder)
    dist1 = Categorical(logits=policy.gate1(feat1))
    a1 = dist1.sample()  # 采样动作 [B]
    logp1 = dist1.log_prob(a1)  # log概率 [B]

    # === 步骤3: 运行初始化器 ===
    sol0 = run_initializers_by_action(coords, a1, init_zoo)

    # === 步骤4: Gate2前向传播 ===
    feat2 = tsp_gate2_features(coords, sol0.tour, sol0.length, cfg=feat_cfg, encoder=policy.encoder)
    dist2 = Categorical(logits=policy.gate2(feat2))
    a2 = dist2.sample()
    logp2 = dist2.log_prob(a2)

    # === 步骤5: 运行迭代器 ===
    sol1 = run_iterators_by_action(coords, a2, sol0, iter_zoo)

    # === 步骤6: 计算奖励 ===
    length = sol1.length
    reward = -length  # 越短越好

    # === 步骤7: 计算Baseline ===
    if cfg.baseline == "batch_mean":
        baseline_length = length.mean()
        baseline_reward = -baseline_length
        baseline1 = baseline_reward
        baseline2 = baseline_reward
    elif cfg.baseline == "critic":
        baseline1 = policy.value1(feat1).squeeze(-1)
        baseline2 = policy.value2(feat2).squeeze(-1)
        critic_loss1 = ((reward - baseline1) ** 2).mean()
        critic_loss2 = ((reward - baseline2) ** 2).mean()
        critic_loss = 0.5 * (critic_loss1 + critic_loss2)
    elif cfg.baseline == "critic_batch_mean":
        baseline_length = length.mean()
        baseline_reward = -baseline_length
        value1_pred = policy.value1(feat1).squeeze(-1)
        value2_pred = policy.value2(feat2).squeeze(-1)
        baseline1 = baseline_reward + value1_pred
        baseline2 = baseline_reward + value2_pred
        critic_loss1 = ((reward - baseline_reward - value1_pred) ** 2).mean()
        critic_loss2 = ((reward - baseline_reward - value2_pred) ** 2).mean()
        critic_loss = 0.5 * (critic_loss1 + critic_loss2)

    # === 步骤8: 计算优势函数 ===
    adv1 = reward - baseline1  # [B]
    adv2 = reward - baseline2  # [B]

    # === 步骤9: 计算Actor损失 ===
    actor_loss = -(logp1 * adv1.detach() + logp2 * adv2.detach()).mean()
    loss = actor_loss + cfg.critic_coef * critic_loss

    # === 步骤10: 反向传播和优化 ===
    optimizer.zero_grad(set_to_none=True)
    loss.backward()
    optimizer.step()
```

### 3.2 三种Baseline模式对比

| 模式 | Baseline计算 | Critic MSE损失 | 优势 |
|------|-------------|---------------|------|
| `batch_mean` | `-mean(length)` | 无 | 简单，但方差大 |
| `critic` | `V(feat1)`, `V(feat2)` | `(reward - V)^2` | 学习值函数，收敛慢 |
| `critic_batch_mean` | `-mean(length) + V(feat)` | `(reward - (-mean) - V)^2` | 结合两者优点 |

**推荐**：`critic_batch_mean`
- Batch mean提供稳定的baseline中心
- Critic学习实例间的相对差异
- 训练更稳定，收敛更快

### 4.1 评估模式

**文件**：`train_two_gate_tsp.py:695-742`

```python
def evaluate(cfg, policy, init_zoo, iter_zoo, feat_cfg, eval_coords):
    """
    评估模式：贪心选择（argmax）

    与训练模式的区别:
    - 不采样，直接选择logits最大的动作
    - 使用固定的随机种子（确保可重复性）
    - 不计算梯度
    """
    policy.eval()

    with torch.no_grad():
        with _TempSeed(cfg.eval_seed):  # 固定随机种子
            for start in range(0, int(eval_coords.size(0)), int(cfg.batch_size)):
                coords = eval_coords[start : start + int(cfg.batch_size)].to(cfg.device)

                # Gate1: 贪心选择
                logits1 = policy.gate1(tsp_instance_features(coords, ...))
                a1 = logits1.argmax(dim=1)  # [B]
                sol0 = run_initializers_by_action(coords, a1, init_zoo)

                # Gate2: 贪心选择
                logits2 = policy.gate2(tsp_gate2_features(coords, sol0.tour, sol0.length, ...))
                a2 = logits2.argmax(dim=1)
                sol1 = run_iterators_by_action(coords, a2, sol0, iter_zoo)

                total_len += float(sol1.length.sum().item())
                total_count += int(sol1.length.numel())

                # 统计动作分布
                a1_counts += torch.bincount(a1.detach().cpu(), ...)
                a2_counts += torch.bincount(a2.detach().cpu(), ...)

    mean_len = total_len / max(total_count, 1)

    return {"mean_len": float(mean_len), "gate1_dist": p1_str, "gate2_dist": p2_str}
```

### 5.1 临时种子上下文管理器

**文件**：`train_two_gate_tsp.py:165-202`

```python
class _TempSeed:
    """
    临时设置随机种子，用于可重复的评估

    用途:
    - 确保评估过程中迭代器（如RRC）的随机性可控
    - 评估完成后恢复原始RNG状态
    """
    def __init__(self, seed: int):
        self.seed = int(seed)

    def __enter__(self):
        # 保存当前状态
        self._py_state = random.getstate()
        self._np_state = np.random.get_state()
        self._torch_state = torch.random.get_rng_state()
        self._cuda_state = torch.cuda.get_rng_state_all() if torch.cuda.is_available() else None

        # 设置新种子
        random.seed(self.seed)
        np.random.seed(self.seed)
        torch.manual_seed(self.seed)
        if torch.cuda.is_available():
            torch.cuda.manual_seed_all(self.seed)
        return self

    def __exit__(self, exc_type, exc, tb):
        # 恢复原始状态
        random.setstate(self._py_state)
        np.random.setstate(self._np_state)
        torch.random.set_rng_state(self._torch_state)
        if self._cuda_state is not None:
            torch.cuda.set_rng_state_all(self._cuda_state)
        return False
```

### 5.2 Torch默认配置

**文件**：`train_two_gate_tsp.py:113-133`

```python
def _configure_torch_defaults(device: str) -> None:
    """
    配置全局torch默认值

    问题:
    - EasyNCO的某些组件创建张量时不指定device
    - 如果运行在CUDA但默认是CPU，会导致设备不匹配错误

    解决:
    - 设置默认tensor类型为torch.cuda.FloatTensor
    """
    import torch

    if device.startswith("cuda"):
        if not torch.cuda.is_available():
            raise RuntimeError(f"Requested device='{device}' but CUDA is not available.")
        if ":" in device:
            _, idx = device.split(":", 1)
            if idx.strip() != "":
                torch.cuda.set_device(int(idx))
        torch.set_default_tensor_type(torch.cuda.FloatTensor)  # 关键！
    else:
        torch.set_default_tensor_type("torch.FloatTensor")
```

**影响**：
- 设置后，所有未指定设备的张量操作都默认在CUDA上
- 包括`torch.randn()`、`torch.zeros()`等

### 5.3 批量内不同动作的处理

```python
# 问题：一个batch中的每个实例可能选择不同的求解器
# 解决：按动作分组，分别调用求解器

def run_initializers_by_action(coords, actions, zoo):
    """
    假设:
    - actions = [0, 1, 0, 2, 1, 0, ...]  # 每个实例选择的初始化器索引

    流程:
    1. 按动作分组: idx_0=[0,2,5,...], idx_1=[1,4,...], idx_2=[3,...]
    2. 对每组调用对应的求解器:
       zoo[0].solve(coords[idx_0])
       zoo[1].solve(coords[idx_1])
       zoo[2].solve(coords[idx_2])
    3. 合并结果
    """
    groups = _gather_by_action(actions, len(zoo))
    for action_id, idx in enumerate(groups):
        if idx.numel() == 0:
            continue
        sub = zoo[action_id].solve(coords[idx])
        out_tour[idx] = sub.tour
        out_len[idx] = sub.length
```
